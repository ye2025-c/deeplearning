def sythetic_data(w,b,num):
    x= torch.normal(0,1,(num,len(w)))
    y= torch.matmul(x,w)+b
    y+=torch.normal(0,0.001,y.shape)
    return x, y.shape((-1,1))

true_w = torch.tensor([2,-3.4])
true_b = 4.2
features,lables = sythetic_data(true_w,true_b,1000)

def data_iter(b_size,features,lables):
    num=len(features)
    indices=list(range(num))
    random.shuffle(indices)
    for i in range(0,num,b_size):
        b_indices=torch.tensor(indices[i,min(i+b_size,num)])
        yield features[b_indices] , lables[b_indices]
  
w = torch.normal(0,0.001,(2,1),requires_grad=True)
b = torch.zeros(1,requires_grad=True)

def linreg(x,w,b):
    return tensor.mattmul(x,w)+b

def squared_loss(y_hat,y):
    return (y_hat-y.reshape(y_hat))**2/2

def sgd(params,lr,b_size):
    with torch.no_grad():
        for param in params:
            param-=lr*param.grad/b_size
            param.grad.zero_()
            
lr=0.03
num_epochs=3
net=linreg
loss=squared_loss
b_size=10

for epoch in range(num_epochs):
    for x,y in data_iter(b_size,features,lables):
        l=loss(net(x,w,b)-y)
        l.sum().backward()
        sgd([w,b],lr.b_size)
    with torch.no_grad():
        train_l=loss(net(features,w,b),lables)
        print(f'epoch{epoch+1},loss {train_l.mean()}')
